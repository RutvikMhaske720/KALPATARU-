section .bss

    char_buff resb 17
    actl      resq 1

    a         resq 1
    b         resq 1
    c         resq 1
    d         resq 1

    choice    resb 2


section .text

    global _start


_start:

    ; Input first number
    WRITE msg1,len1

    call accept
    mov qword[a],rbx

    ; Input second number
    call accept
    mov qword[b],rbx


printmenu:

    ; Display menu
    WRITE menu,menulen

    ; Read choice
    READ choice,2

    cmp byte[choice],31H       ; 1
    je addition

    cmp byte[choice],32H       ; 2
    je subtraction

    cmp byte[choice],33H       ; 3
    je multiplication

    cmp byte[choice],34H       ; 4
    je division

    cmp byte[choice],35H       ; 5
    je exit

    ; Invalid choice
    WRITE msg7,len7
    jmp printmenu


; -----------------------------------
; ADDITION
; -----------------------------------

addition:

    mov rbx,qword[a]
    add rbx,qword[b]
    mov qword[c],rbx

    WRITE msg2,len2

    mov rbx,qword[c]
    call display

    jmp printmenu


; -----------------------------------
; SUBTRACTION
; -----------------------------------

subtraction:

    mov rbx,qword[a]
    sub rbx,qword[b]
    mov qword[c],rbx

    WRITE msg3,len3

    mov rbx,qword[c]
    call display

    jmp printmenu


; -----------------------------------
; MULTIPLICATION
; -----------------------------------

multiplication:

    mov rax,qword[a]
    mul qword[b]

    ; RDX:RAX contains the result
    mov qword[c],rdx
    mov qword[d],rax

    WRITE msg4,len4

    ; Display upper 64 bits
    mov rbx,qword[c]
    call display

    ; Display lower 64 bits
    mov rbx,qword[d]
    call display

    jmp printmenu


; -----------------------------------
; DIVISION
; -----------------------------------

division:

    ; Check if divisor is zero
    cmp qword[b],0
    je divide_zero

    mov rdx,0
    mov rax,qword[a]

    div qword[b]

    ; RAX = quotient
    ; RDX = remainder

    mov qword[c],rax
    mov qword[d],rdx

    WRITE msg5,len5

    mov rbx,qword[c]
    call display

    WRITE msg6,len6

    mov rbx,qword[d]
    call display

    jmp printmenu


; -----------------------------------
; DIVISION BY ZERO
; -----------------------------------

divide_zero:

    WRITE msg8,len8
    jmp printmenu


; -----------------------------------
; EXIT
; -----------------------------------

exit:

    EXIT


; -----------------------------------
; ACCEPT HEX NUMBER
; -----------------------------------

accept:

    READ char_buff,17

    dec rax
    mov qword[actl],rax

    mov rbx,00H
    mov rsi,char_buff


up:

    shl rbx,04H

    mov rdx,00H
    mov dl,byte[rsi]

    cmp dl,39H
    jbe sub30

    sub dl,07H


sub30:

    sub dl,30H
    add rbx,rdx

    inc rsi

    dec qword[actl]
    jnz up

    ret


; -----------------------------------
; DISPLAY HEX NUMBER
; -----------------------------------

display:

    mov rsi,char_buff
    mov rcx,16


above:

    rol rbx,04H

    mov dl,bl
    and dl,0FH

    cmp dl,09H
    jbe add30

    add dl,07H


add30:

    add dl,30H

    mov byte[rsi],dl

    inc rsi

    dec rcx
    jnz above

    WRITE char_buff,16

    Ret

; SUCCESSIVE ADDITION METHOD

succadd:

    mov rbx,00H
    mov rcx,[n]

    cmp rcx,00H
    je printmenu

l1:
    add rbx,[m]
    dec rcx
    jnz l1

    mov [ans],rbx

    WRITE msg4,len4

    mov rbx,[ans]
    call display

    jmp printmenu


; SHIFT AND ADD METHOD

shiftadd:

    mov qword [A],00H

    mov rbx,[m]
    mov [B],rbx

    mov rbx,[n]
    mov [Q],rbx

    mov rcx,64

l2:
    mov rbx,[Q]
    and rbx,01H
    jz shiftaq

    mov rbx,[B]
    add [A],rbx

shiftaq:

    shr qword [Q],01H

    mov rbx,[A]
    and rbx,01H
    jz shifta

    mov rbx,01H
    shl rbx,63
    or qword [Q],rbx

shifta:

    shr qword [A],01H

    dec rcx
    jnz l2

    WRITE msg4,len4

    mov rbx,[A]
    call display

    mov rbx,[Q]
    call display

    jmp printmenu

exit:
    EXIT

section .bss

char_buff resb 16

section .text
global _start

_start:
    WRITE msg1, len1
    READ str1, 30
    dec rax
    mov [l1], rax

printmenu:
    WRITE menu, menulen
    READ choice, 2

    cmp byte[choice], 31H
    je strlen
    cmp byte[choice], 32H
    je strcpy
    cmp byte[choice], 33H
    je strcat
    cmp byte[choice], 34H
    je strrev
    cmp byte[choice], 35H
    je strcmp
    cmp byte[choice], 36H
    je Pal
    cmp byte[choice], 37H
    je substr
    cmp byte[choice], 38H
    je exit

    WRITE msg3, len3
    jmp printmenu

strlen:
    WRITE msg4, len4
    mov rbx, [l1]
    call display
    jmp printmenu

strcpy:
    mov rsi, str1
    mov rdi, str3
    mov rcx, [l1]
    CLD
    rep movsb

    WRITE msg5, len5
    WRITE str3, [l1]
    jmp printmenu

strcat:
    WRITE msg2, len2
    READ str2, 30
    dec rax
    mov [l2], rax

    mov rsi, str1
    mov rdi, str3
    mov rcx, [l1]
    CLD
    rep movsb

    mov rsi, str2
    mov rcx, [l2]
    rep movsb

    mov rax, [l1]
    add rax, [l2]
    mov [l3], rax

    WRITE msg6, len6
    WRITE str3, [l3]
    jmp printmenu

strrev:
    mov rsi, str1
    mov rdi, str3
    add rdi, [l1]
    dec rdi
    mov rcx, [l1]
    CLD

revloop:
    cmp rcx, 00
    je revdone

    mov dl, byte[rsi]
    mov byte[rdi], dl
    inc rsi
    dec rdi
    dec rcx
    jmp revloop

revdone:
    WRITE msg7, len7
    WRITE str3, [l1]
    jmp printmenu

strcmp:
    WRITE msg2, len2
    READ str2, 30
    dec rax
    mov [l2], rax

    mov rax, [l1]
    cmp rax, [l2]
    jne ntequal

    mov rsi, str1
    mov rdi, str2
    mov rcx, [l1]
    CLD
    repe cmpsb
    jne ntequal

    WRITE msg8, len8
    jmp printmenu

ntequal:
    WRITE msg9, len9
    jmp printmenu

Pal:
    mov rsi, str1
    mov rdi, str1
    add rdi, [l1]
    dec rdi
    mov rcx, [l1]
    shr rcx, 1

palloop:
    cmp rcx, 00
    je palyes

    mov al, byte[rsi]
    cmp al, byte[rdi]
    jne palno

    inc rsi
    dec rdi
    dec rcx
    jmp palloop

palyes:
    WRITE msg10, len10
    jmp printmenu

palno:
    WRITE msg11, len11
    jmp printmenu

substr:
    WRITE msg12, len12
    READ str2, 30
    dec rax
    mov [l2], rax

    mov r8, [l1]
    mov r9, [l2]

    cmp r9, 00
    je subfound

    cmp r9, r8
    ja subnotfound

    mov r10, 00

subloop:
    mov rsi, str1
    add rsi, r10
    mov rdi, str2
    mov rcx, r9
    CLD
    repe cmpsb
    je subfound

    inc r10
    mov rax, r8
    sub rax, r9
    cmp r10, rax
    jbe subloop

subnotfound:
    WRITE msg14, len14
    jmp printmenu

subfound:
    WRITE msg13, len13
    jmp printmenu

display:
    mov rsi, char_buff
    mov rcx, 16

above:
    rol rbx, 04H
    mov dl, bl
    AND dl, 0FH
    cmp dl, 09H
    jbe add30
    add dl, 07H

add30:
    add dl, 30H
    mov byte[rsi], dl
    inc rsi
    dec rcx
    jnz above

    WRITE char_buff, 16
    RET

exit:
    EXIT

extern printf, scanf

%macro PRINT 02
  push rbp
  mov rax, 00
  mov rdi, %1
  mov rsi, %2
  call printf
  pop rbp
%endmacro

%macro SCAN 02
  push rbp
  mov rax, 00
  mov rdi, %1
  mov rsi, %2
  call scanf
  pop rbp
%endmacro

%macro PRINTFLOAT 02
  push rbp
  mov rax, 01
  mov rdi, %1
  movsd xmm0, %2
  call printf
  pop rbp
%endmacro

section .data
  msg1 db "Enter a, b and c: ", 0
  msg2 db "Roots are: ", 0
  msg3 db "Complex roots due to negative Discriminant", 10
  fmt1 db "%lf", 0
  fmt2 db "%s", 0

section .bss
  a resq 1
  b resq 1
  c resq 1
  t1 resq 1
  t2 resq 1
  t3 resq 1
  t4 resq 1
  temp resw 1
  r1 resq 1
  r2 resq 1


section .text
  global main
  
main:
  PRINT fmt2, msg1
  SCAN fmt1, a
  SCAN fmt1, b
  SCAN fmt1, c
  FINIT
  
  fld qword[b]
  fmul qword[b]
  fstp qword[t1]	;t1 = b^2
  
  fld qword[a]
  fmul qword[c]
  mov word[temp], 04
  fimul word[temp]
  fstp qword[t2]	;t2 = 4ac
  
  fld qword[t1]
  fsub qword[t2]
  fstp qword[t1] 	;t1 = b2 - 4ac
  
  fld qword[t1]
  fabs
  fsqrt
  fstp qword[t2]	;t2 = root of D
  
  fld qword[a]
  mov word[temp], 02
  fimul word[temp]
  fstp qword[t4]	;t4 = 2a
  
  PRINT fmt2, msg2
  cmp qword[t1], 00H	; t1 = D
  je printsecond
  jl complexroot
  fld qword[b]
  fchs
  fadd qword[t2]
  fdiv qword[t4]  
  fstp qword[r1]
  PRINTFLOAT fmt1, [r1]

printsecond:
  fld qword[b]
  fchs
  fsub qword[t2]
  fdiv qword[t4]  
  fstp qword[r2]
  PRINTFLOAT fmt1, [r2]
  mov rax, 00
  RET
  
complexroot:
  PRINT fmt2, msg3
  mov rax, 00
  RET
