A = [[2, 3], [4, 5]]
B = [[1, 0], [2, 1]]


def addition(A, B):
    return [[A[i][j] + B[i][j] for j in range(2)] for i in range(2)]


def multiplication(A, B):
    C = [[0, 0], [0, 0]]

    for i in range(2):
        for j in range(2):
            for k in range(2):
                C[i][j] += A[i][k] * B[k][j]

    return C


def scalar(A, n):
    return [[A[i][j] * n for j in range(2)] for i in range(2)]


print("Matrix A:", A)
print("Matrix B:", B)
print("Addition:", addition(A, B))
print("Multiplication:", multiplication(A, B))
print("Scalar Multiplication:", scalar(A, 3))

              [2, 3, 0],

print("Original Matrix A:")
print(A)

print(vectors)

P_inv = np.linalg.inv(P)

print("\nDiagonal Matrix D:")
print(D)


print("\nReconstructed Matrix (P * D * P inverse):")
print(np.round(P @ D @ P_inv, 2))

print("\nA^5 using Direct Method:")
print(np.round(A1, 2))

import numpy as np
from scipy.stats import norm

def z_test(data, mu):
    n = len(data)
    x = np.mean(data)
    s = np.std(data, ddof=1)

    z = (x - mu) / (s / np.sqrt(n))
    p = 2 * (1 - norm.cdf(abs(z)))

    print("n                  :", n)
    print("Sample Mean        :", round(x, 4))
    print("Hypothesized Mean  :", mu)
    print("Z-statistic        :", round(z, 4))
    print("p-value            :", round(p, 6))
    print("alpha              :", 0.05)

    if p <= 0.05:
        print("Conclusion         : Reject H0")
        print("\nWe REJECT the claim that mean fill volume is 50 ml.")
    else:
        print("Conclusion         : Fail to Reject H0")
        print("\nWe DO NOT REJECT the claim that mean fill volume is 50 ml.")


np.random.seed(1)
data = np.random.normal(50.6, 2.0, 40)

z_test(data, 50)

import numpy as np
from scipy.stats import f

def f_test(A, B):
    v1 = np.var(A, ddof=1)
    v2 = np.var(B, ddof=1)

    n1 = len(A)
    n2 = len(B)

    if v1 >= v2:
        F = v1 / v2
        df1 = n1 - 1
        df2 = n2 - 1
    else:
        F = v2 / v1
        df1 = n2 - 1
        df2 = n1 - 1

    p = 2 * min(f.cdf(F, df1, df2), 1 - f.cdf(F, df1, df2))
    critical = f.ppf(0.975, df1, df2)

    print("Variance 1       :", round(v1, 4))
    print("Variance 2       :", round(v2, 4))
    print("F-statistic      :", round(F, 4))
    print("df1, df2         :", (df1, df2))
    print("Critical F       :", round(critical, 4))
    print("p-value          :", round(p, 6))

    if F > critical:
        print("Conclusion       : Reject H0 (variances differ)")
    else:
        print("Conclusion       : Fail to Reject H0 (variances equal)")


A = [10.2, 9.8, 10.5, 10.1, 9.9, 10.3, 10.0]
B = [9.5, 10.6, 9.2, 10.8, 9.4, 10.9, 9.1]

f_test(A, B)

import numpy as np
import matplotlib.pyplot as plt


def validate(P):
    if P.shape[0] != P.shape[1] or not np.allclose(P.sum(axis=1), 1):
        raise ValueError("Invalid Matrix")
    return P


def transition(P, n):
    return np.linalg.matrix_power(validate(P), n)


P = np.array([
    [0.7, 0.2, 0.1],
    [0.1, 0.6, 0.3],
    [0.4, 0.2, 0.4]
])

names = ["Sunny", "Cloudy", "Rainy"]


for n in [1, 2, 5, 10, 20]:
    print("\nP^", n, ":\n", np.round(transition(P, n), 4))


for i in range(3):
    values = [transition(P, n)[0][i] for n in range(1, 21)]
    plt.plot(range(1, 21), values, marker="o", label=names[i])

plt.xlabel("Steps")
plt.ylabel("Probability")
plt.title("Transition Probability from Sunny")
plt.legend()
plt.grid()
plt.show()

import numpy as np

# Historical Weather Data
weather = [
    "Sunny", "Cloudy", "Rainy", "Rainy",
    "Cloudy", "Sunny", "Cloudy", "Rainy",
    "Sunny", "Sunny", "Cloudy", "Rainy",
    "Rainy", "Cloudy", "Sunny", "Cloudy"
]


counts = {}

for i in range(2, len(weather)):
    pair = (weather[i-2], weather[i-1])
    next_state = weather[i]

    if pair not in counts:
        counts[pair] = {}

    if next_state not in counts[pair]:
        counts[pair][next_state] = 0

    counts[pair][next_state] += 1


print("Learned transition probabilities:")

for pair, states in counts.items():
    total = sum(states.values())
    print("\n", pair, "->")

    for state, count in states.items():
        p = count / total
        print(" ", state, ":", round(p * 100, 2), "%")


current = (weather[-2], weather[-1])

if current in counts:
    states = list(counts[current].keys())
    total = sum(counts[current].values())

    probabilities = [counts[current][s] / total for s in states]

    rng = np.random.default_rng(42)
    prediction = rng.choice(states, p=probabilities)

    print("\nPrevious two states:", current)
    print("Predicted next state:", prediction)
else:
    print("Insufficient data for this pair.")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score


np.random.seed(42)

bathrooms = np.random.randint(1, 4, n)

    "Size_sqft": size,
    "Bedrooms": bedrooms,
    "Bathrooms": bathrooms,
    "Distance_km": distance,

import numpy as np
import matplotlib.pyplot as plt

x = [1, 2, 3, 4, 5]
y = [2, 4, 1, 3, 7]
x_new = 2.5


def lagrange(x, y, value):
    result = 0

    for i in range(len(x)):
        term = y[i]

        for j in range(len(x)):
            if i != j:
                term = term * (value - x[j]) / (x[i] - x[j])

        result = result + term

    return result


y_new = lagrange(x, y, x_new)

print("Interpolated value:", round(y_new, 4))


x_range = np.linspace(min(x), max(x), 200)
y_range = [lagrange(x, y, i) for i in x_range]

plt.scatter(x, y, color="red", label="Known points")
plt.plot(x_range, y_range, color="blue", label="Interpolation curve")
plt.scatter(x_new, y_new, color="green", marker="x", s=100,
            label="Interpolated point")

plt.title("Lagrange Interpolation")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.grid()
plt.show()

import numpy as np
import matplotlib.pyplot as plt

x = [1, 2, 3, 4]
y = [1, 4, 9, 16]

def divided_diff(x, y):
    n = len(x)
    table = [[0]*n for i in range(n)]

    for i in range(n):
        table[i][0] = y[i]

    for j in range(1, n):
        for i in range(n-j):
            table[i][j] = (table[i+1][j-1] - table[i][j-1]) / (x[i+j] - x[i])

    return table

def newton(x, y, value):
    table = divided_diff(x, y)
    result = table[0][0]
    product = 1

    for i in range(1, len(x)):
        product *= (value - x[i-1])
        result += table[0][i] * product

    return result

table = divided_diff(x, y)

for row in table:
    print(["%.4f" % v for v in row])

value = 2.5
answer = newton(x, y, value)
print("Interpolated value:", round(answer, 4))

x_range = np.linspace(min(x), max(x), 200)
y_range = [newton(x, y, i) for i in x_range]

plt.scatter(x, y, color="red", label="Known points")
plt.plot(x_range, y_range, color="purple", label="Newton curve")
plt.scatter(value, answer, color="green", marker="x", s=100)

plt.title("Newton's Divided-Difference Interpolation")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.grid()
plt.show()
